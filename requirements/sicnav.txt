# SICNav baseline only -- a SEPARATE environment from requirements/main.txt.
#
#   conda create -n sicnav python=3.8 && conda activate sicnav
#   pip install -r requirements/sicnav.txt
#   pip install -e crowdnav_env
#   pip install -e baselines/sicnav
#
# Why separate: SICNav's MPC solves through CasADi's IPOPT plugin, which
# resolves reliably only against the conda-forge IPOPT/CasADi build on
# Python 3.8. Attempting to co-install it with the main environment's
# torch 2.6 / numpy 1.26 stack breaks the plugin lookup at solve time.
# Every SICNav evaluation therefore runs under its own interpreter; the
# evaluation harness itself is environment-agnostic.
#
# Install IPOPT from conda-forge BEFORE pip:
#   conda install -c conda-forge ipopt casadi=3.6.5
